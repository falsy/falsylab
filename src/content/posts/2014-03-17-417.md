    placeholderTarget.focus(function(){
        $(this).siblings(&quot;label&quot;).fadeOut(&quot;fast&quot;);
    });

    placeholderTarget.focusout(function(){
        if($(this).val() == &quot;&quot;){
            $(this).siblings(&quot;label&quot;).fadeIn(&quot;fast&quot;);
        }
    });

        <p class="left-line-box">
          ECMAScript 2015(ES6)에 추가된 기본 문법들을 간단한 코드를 통해 알아보려 합니다. (설명이 아닌 그냥 가볍게 훑어 볼 수 있는 코드입니다.)
        </p>
        <h3>1. 블록 스코프</h3>
 
<pre class="legacy-code"><code class="language-javascript">var a = &#x27;a&#x27;;
var a = &#x27;b&#x27;;
console.log(a);
// &#x27;b&#x27;

let a = &#x27;a&#x27;;
let a = &#x27;b&#x27;;
// SyntaxError</code></pre> 
 
<pre class="legacy-code"><code class="language-javascript">var a = &#x27;a&#x27;;
if(true) {
  var a = &#x27;b&#x27;;
  console.log(a);
  // &#x27;b&#x27;
}
console.log(a);
// &#x27;b&#x27;

let a = &#x27;a&#x27;;
if(true) {
  let a = &#x27;b&#x27;;
  console.log(a);
  // &#x27;b&#x27;
}
console.log(a);
// &#x27;a&#x27;</code></pre> 

 
<pre class="legacy-code"><code class="language-javascript">let a = &#x27;a&#x27;;
a = &#x27;b&#x27;;
console.log(a);
// &#x27;b&#x27;

const a = &#x27;a&#x27;;
a = &#x27;b&#x27;;
// TypeError</code></pre> 
 
<pre class="legacy-code"><code class="language-javascript">const a = { a: &#x27;a&#x27; }
a.a = &#x27;b&#x27;;
console.log(a);
// {a: &#x27;b&#x27;}

a = { b: &#x27;b&#x27; };
// TypeError</code></pre> 


        <h3>2. 파라미터 기본값</h3>
 
<pre class="legacy-code"><code class="language-javascript">const a = (a = false) =&gt; {
  console.log(a);
};
a(true);
// true;
a();
// false;</code></pre> 

        <h3>3. 펼침 연산자,  나머지 파라미터</h3>
 
<pre class="legacy-code"><code class="language-javascript">// 펼침 연산자
let arr = [1, 2];
arr = [...arr, 3, 4, ...[5, 6]];
console.log(arr);
// [1, 2, 3, 4, 5, 6]

const sum = (a, b, c, d) =&gt; {
  return a+b+c+d;
};
sum(...arr);
// 10

// 나머지 파라미터
const arg = (a, b, ...arg) =&gt; {
  console.log(arg);
};
arg(...arr);
// [3, 4, 5, 6]</code></pre> 

        <h3>4. 해체 할당</h3>
 
<pre class="legacy-code"><code class="language-javascript">let arr = [1, 2, 3, 4];
[a, b, c, d] = arr;
console.log(b);
// 2

[, , , d] = arr;
console.log(d);
// 4</code></pre> 
 
<pre class="legacy-code"><code class="language-javascript">let obj = { a: &#x27;a&#x27;, b: &#x27;b&#x27; };
let {a, b} = obj
console.log(b);
// &#x27;b&#x27;</code></pre> 

        <h3>5. 화살표 함수</h3>
 
<pre class="legacy-code"><code class="language-javascript">let a = (a) =&gt; {
  console.log(a);
}
// 한줄구문의 경우 {} 를 생략할 수 있다.
let a = (a) =&gt; console.log(a);
// 아규먼트가 하나인 경우 ()를 생략할 수 있다.
let a = a =&gt; console.log(a);</code></pre> 

        <p class="basic-text-box">
          화살표 함수는 구문이 짧아진 것 이외에도 일반 함수와 this가 가르키는 오브제의 규칙이 다르며, new 키워드를 사용하여 객체 생성자로 사용할 수 없습니다.
        </p>

        <h3>6. 객체 리터널의 프로퍼티 및 메소드 정의</h3>
 
<pre class="legacy-code"><code class="language-javascript">let b = &#x27;bbb&#x27;;
let obj = {
  a: &#x27;aaa&#x27;,
  b
};
console.log(obj);
// {a: &quot;aaa&quot;, b: &quot;bbb&quot;}</code></pre> 
 
<pre class="legacy-code"><code class="language-javascript">let obj = {
  [&#x27;aa&#x27; + &#x27;bb&#x27;] : &#x27;ccc&#x27;
};
console.log(obj);
// {aabb: &quot;ccc&quot;}</code></pre> 
 
<pre class="legacy-code"><code class="language-javascript">let obj = {
  sum(a, b) {
    return a + b;
  }
};
console.log(obj.sum(1, 2));
// 3</code></pre>

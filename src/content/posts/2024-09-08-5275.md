<h3>1. TCP Server</h3>
<p class="basic-text-box">
  간단하게 Node.js로 TCP 서버를 만들고 실행해줍니다.
</p>

<pre class="legacy-code"><code class="language-javascript">// tcp-server/index.js
const net = require(&quot;net&quot;)

const tcpServer = net.createServer((socket) =&gt; {
  socket.on(&quot;data&quot;, (data) =&gt; {
    console.log(`TCP 요청: ${data}`)
    socket.write(`TCP 응답: ${data}`)
  })

  socket.on(&quot;close&quot;, () =&gt; {
    console.log(&quot;TCP 중단&quot;)
  })
})

tcpServer.listen(8000, &quot;localhost&quot;, () =&gt; {
  console.log(&quot;TCP 서버가 포트 8000에 실행 중입니다.&quot;)
})</code></pre>

<pre class="legacy-code"><code class="language-bash">  $ node ./tcp-server/index.js</code></pre>

<p class="basic-text-box">
  아주 간단하게 8000번 포트에 서버를 열고 요청이 오면 그 데이터를 그대로 응답하는 코드로 되어 있습니다.
</p>

<h3>2. UDP Server</h3>
<p class="basic-text-box">
  Node.js로 UDP 서버도 간단하게 만들고 실행해줍니다.
</p>

<pre class="legacy-code"><code class="language-javascript">// udp-server/index.js
const dgram = require(&quot;dgram&quot;)

const udpServer = dgram.createSocket(&quot;udp4&quot;)

udpServer.on(&quot;message&quot;, (msg, rinfo) =&gt; {
  console.log(`UDP 요청: ${msg}`)
  const response = `UDP 응답: ${msg}`
  udpServer.send(response, rinfo.port, rinfo.address, (err) =&gt; {
    if (err) console.error(&quot;UDP 전송 오류:&quot;, err)
  })
})

udpServer.bind(7000, () =&gt; {
  console.log(&quot;UDP 서버가 포트 7000에 실행 중입니다.&quot;)
})</code></pre>

<pre class="legacy-code"><code class="language-bash">  $ node ./udp-server/index.js</code></pre>

<p class="basic-text-box">
  UDP 서버는 7000번 포트에 실행해 주었습니다.

<h3>4. 응답 속도</h3>

<p class="basic-text-box">
  이제 본격적으로 TCP와 UDP의 차이를 확인해볼까요?
</p>

<p class="basic-text-box">
  우선 속도를 확인할 수 있도록 WebSocket 서버에서 TCP와 UDP서버에 메시지를 전송하고 응답받는 속도를 확인해 볼게요.
</p>

<pre class="legacy-code"><code class="language-javascript">// websocket-server/index.js
const WebSocket = require(&quot;ws&quot;)
const net = require(&quot;net&quot;)
const dgram = require(&quot;dgram&quot;)

const wss = new WebSocket.Server({ port: 8080 })

const tcpClient = new net.Socket()
tcpClient.connect(8000, &quot;localhost&quot;, () =&gt; {
  console.log(&quot;TCP 서버와 연결&quot;)
})

const udpClient = dgram.createSocket(&quot;udp4&quot;)

let startTime = 0

// TCP 서버로부터 응답 수신
tcpClient.on(&quot;data&quot;, (data) =&gt; {
  const endTime = Date.now()
  console.log(`TCP 서버 응답 시간: ${endTime - startTime}ms`)
  wss.clients.forEach((client) =&gt; {
    if (client.readyState === WebSocket.OPEN) {
      client.send(`TCP 서버 응답: ${data}`)
    }
  })
})

// UDP 서버로부터 응답 수신
udpClient.on(&quot;message&quot;, (msg, rinfo) =&gt; {
  const endTime = Date.now()
  console.log(`UDP 서버 응답 시간: ${endTime - startTime}ms`)
  wss.clients.forEach((client) =&gt; {
    if (client.readyState === WebSocket.OPEN) {
      client.send(`UDP 서버 응답: ${msg}`)
    }
  })
})

wss.on(&quot;connection&quot;, (ws) =&gt; {
  console.log(&quot;웹 소캣 연결&quot;)

  ws.on(&quot;message&quot;, (message) =&gt; {
    console.log(`클라이언트에게 받은 메시지: ${message}`)

    startTime = Date.now()

    // TCP 서버에 메시지 전송
    tcpClient.write(message)

    // UDP 서버에 메시지 전송
    udpClient.send(message, 7000, &quot;localhost&quot;, (err) =&gt; {
      if (err) console.error(&quot;UDP 서버에 전송 실패:&quot;, err)
    })
  })
})</code></pre>

<p class="basic-text-box">
  자, 이제 빠르게 메시지를 1, 2, 3 보내봅니다!
</p>

  짠!
</p>

<p class="basic-text-box">
  맙소사. TCP가 UDP보다 빠르네요..
</p>

<p class="basic-text-box">
  생각해보니까, 일반적으로 UDP가 TCP보다 빠르다고 하는 건 다양한 이유가 있지만, 그중에 TCP는 신뢰성 있는 데이터를 보장하기 위해 데이터 전송 전에 동기화 패킷을 주고 받는 검증 절차(3-Way Handshake)가 있기 때문인데, 지금 연결은 WebSocket관 연결하다보니, 처음에 한번 이 절차를 수행하고 연결 후에는 바로 데이터를 전송하기 때문에 TCP가 더 빠른 결과가 나온 거 같아요.
</p>

<p class="basic-text-box">
  TCP 연결은 통신마다 다시 연결하도록 코드를 수정하고 다시 테스트 해볼게요.
</p>

<pre class="legacy-code"><code class="language-javascript">// websocket-server/index.js

      client.send(`UDP 서버 응답: ${msg}`)
    }
  })

  console.log(&quot;웹 소캣 연결&quot;)

  ws.on(&quot;message&quot;, (message) =&gt; {
    console.log(`클라이언트에게 받은 메시지: ${message}`)

      tcpClient.write(message)

<p class="basic-text-box">
  오! 큰 차이는 아니지만 UDP가 TCP 보다 약간 빨라졌어요. 지금은 아주 데이터가 작고 그리고 로컬 환경에서 TCP는 OS에서 더 많이 최적화가 되어 있어서 그렇고, 실제 서비스에서는 아마도 더 차이가 크게 나겠죠?!
</p>

<h3>5. 데이터 무결성</h3>

<p class="basic-text-box">
  TCP의 또 중요한 특징중에 하나는 패킷의 순서 보장인데, 이 패킷의 순서는 애플리케이션단에서는 확인이 어려울 거 같아서, 대신 UDP의 데이터 손실에 대해서 확인을 해보려고 합니다!
</p>

<p class="basic-text-box">
TCP라면, 데이터 전송 중 패킷이 손실이 발생하면 손실된 패킷을 다시 전송해서 무결성을 보장했겠죠?!
</p>

<p class="left-line-box">
  이전 응답 속도 테스트랑 같이 구현하려니 코드가 좀 복잡해져서 코드를 분리해서 따로 만들었어요.
</p>

<pre class="legacy-code"><code class="language-javascript">// image-udp-server/index.js
const dgram = require(&quot;dgram&quot;)

const udpServer = dgram.createSocket(&quot;udp4&quot;)
const CHUNK_SIZE = 6 * 1024 // 6KB

udpServer.on(&quot;message&quot;, (message, rinfo) =&gt; {
  let offset = 0

  while (offset &lt; message.length) {
    const chunk = message.slice(offset, offset + CHUNK_SIZE)

    udpServer.send(chunk, rinfo.port, rinfo.address, (err) =&gt; {
      if (err) {
        console.error(&quot;UDP 전송 오류:&quot;, err)
      } else {
        console.log(`청크 전송 완료: ${chunk.length} bytes`)
      }
    })

    offset += CHUNK_SIZE
  }
})

udpServer.bind(7000, () =&gt; {
  console.log(&quot;UDP 서버가 포트 7000에 실행 중입니다.&quot;)
})</code></pre>

<p class="basic-text-box">
  예전 기억에, MacOS에서 UDP 패킷이 최대 9KB 까지만 가능하다고 본 기억이 있어서 청크의 크기는 적당히 작은 6KB로 짤라서 보내도록 했어요.
</p>

<p class="basic-text-box">
  WebSocket 서버에서도 동일하게 UDP 서버로 6KB 씩 나눠서 보내도록 했어요.
</p>

export default function App() {
  const ws = useRef(null)
  const timeout = useRef(null)
  const chunks = useRef([])
  const [imageFile, setImageFile] = useState(null)
  const [uploadedImageURL, setUploadedImageURL] = useState(null)
  const [receivedImageURL, setReceivedImageURL] = useState(null)

  const handleFileChange = (e) =&gt; {
    const file = e.target.files[0]
    setImageFile(file)

    if (file) {
      const reader = new FileReader()
      reader.onload = (e) =&gt; {
        const url = URL.createObjectURL(file)
        setUploadedImageURL(url)
      }
      reader.readAsDataURL(file)
    }
  }

  const sendImage = () =&gt; {
    if (imageFile) {
      const reader = new FileReader()
      reader.onload = (e) =&gt; {
        const arrayBuffer = e.target.result
        ws.current.send(arrayBuffer)
      }
      reader.readAsArrayBuffer(imageFile)
    }
  }

  useEffect(() =&gt; {
    ws.current = new WebSocket(&quot;ws://localhost:8080&quot;)

    ws.current.onopen = () =&gt; {
      console.log(&quot;connected&quot;)
    }

    ws.current.onmessage = (event) =&gt; {
      // 수신한 Blob 청크를 배열에 추가
      chunks.current = [...chunks.current, event.data]

      // 이전 타이머가 존재하면 클리어
      if (timeout.current) {
        clearTimeout(timeout.current)
      }

      // 새로운 타이머 설정 (1초 후에 결합 시도)
      timeout.current = setTimeout(() =&gt; {
        const blob = new Blob(chunks.current, { type: &quot;image/jpeg&quot; }) // 모든 청크를 합침
        const url = URL.createObjectURL(blob)
        console.log(&quot;received image url:&quot;, url)
        setReceivedImageURL(url)
        chunks.current = [] // 청크 배열 초기화
      }, 1000)
    }

    return () =&gt; {
      if (ws.current) {
        ws.current.close()
      }
    }
  }, [])

  return (
    &lt;div&gt;
      &lt;section&gt;
        &lt;div&gt;
          &lt;h2&gt;UDP 이미지 업로드&lt;/h2&gt;
        &lt;/div&gt;
        &lt;div&gt;
          &lt;input type=&quot;file&quot; accept=&quot;image/*&quot; onChange={handleFileChange} /&gt;
          &lt;button onClick={sendImage}&gt;Send Image&lt;/button&gt;
        &lt;/div&gt;
        &lt;div style={{ display: &quot;grid&quot;, gridTemplateColumns: &quot;1fr 1fr&quot; }}&gt;
          &lt;div&gt;
            &lt;h2&gt;업로드 이미지:&lt;/h2&gt;
            {uploadedImageURL &amp;&amp; (
              &lt;img
                src={uploadedImageURL}
                style={{ maxWidth: &quot;300px&quot;, maxHeight: &quot;300px&quot; }}
              /&gt;
            )}
          &lt;/div&gt;
          &lt;div&gt;
            &lt;h2&gt;수신 이미지:&lt;/h2&gt;
            {receivedImageURL &amp;&amp; (
              &lt;img
                src={receivedImageURL}
                style={{ maxWidth: &quot;300px&quot;, maxHeight: &quot;300px&quot; }}
              /&gt;
            )}
          &lt;/div&gt;
        &lt;/div&gt;
      &lt;/section&gt;
    &lt;/div&gt;
  )
}</code></pre>

<p class="view_box_title" style="margin-top: 30px;">Explanation</p>

<div id="review_text">
<p class="basic-text-box">구글 캘린더 API를 활용하여 구글 사용자 인증과 캘린더의 이벤트를 받아오고 캘린더를 추가 및 삭제하는 방법입니다. 내용을 구글 API 사이트의 'Quickstart'를 기반으로 하였습니다.
참고 : <a href="https://developers.google.com/google-apps/calendar/quickstart/nodejs#prerequisites" target="_blank">구글 Quickstart (node.js)</a></p>

<h3>1. 시작하기전에</h3>
<p class="basic-text-box">node.js 와 npm이 설치 되어있다면</p>

<pre class="legacy-code"><code class="language-javascript">npm install googleapis --save
npm install google-auth-library --save</code></pre>
<p class="basic-text-box">위와 같이 클라이언트 라이브러리를 설치합니다.
그리고 구글에 로그인한 후 <a href="https://console.developers.google.com/flows/enableapi?apiid=calendar" target="_blank">https://console.developers.google.com/flows/enableapi?apiid=calendar</a>로 이동하여 API 프로젝트를 생성하고 API키를 발급 받습니다.</p>

<h3>2. 시작</h3>
<p class="basic-text-box">프로젝트를 생성하여 API를 발급 받은 후 '사용자 인증 정보' 페이지의 'OAuth 2.0 클라이언트 ID' 가장 오른쪽에 있는 다운로드 아이콘을 클릭하여 'client_secret_xxxx.json' 다운로드 합니다. 그리고 파일 이름을 'client_secret.json'으로 변경하여 node의 디렉토리에 넣습니다. 그리고 아래와 같은  'quickstart.js' 파일을 만들어보겠습니다.</p>

<pre class="legacy-code"><code class="language-javascript">var fs = require(&#x27;fs&#x27;);
var readline = require(&#x27;readline&#x27;);
var google = require(&#x27;googleapis&#x27;);
var googleAuth = require(&#x27;google-auth-library&#x27;);

var SCOPES = [
              &#x27;https://www.googleapis.com/auth/calendar.readonly&#x27;, // 캘린더 정보의 읽기 권한? 입니다.
              &#x27;https://www.googleapis.com/auth/calendar&#x27; // 캘린더 정보의 편집 권한? 입니다.
              ];

var TOKEN_DIR = (process.env.HOME || process.env.HOMEPATH ||
    process.env.USERPROFILE) + &#x27;/.credentials/&#x27;;
// 토큰이 저장될 디렉토리
var TOKEN_PATH = TOKEN_DIR + &#x27;calendar-nodejs-quickstart.json&#x27;;
// 토큰 파일 명

fs.readFile(&#x27;client_secret.json&#x27;, function processClientSecrets(err, content) {
  // 로컬의 client_secret.json 파일을 읽고 &#x27;authorize&#x27; 함수 실행
  if (err) {
    console.log(&#x27;Error loading client secret file: &#x27; + err);
    return;
  }
  authorize(JSON.parse(content), listEvents);
});

function authorize(credentials, callback) {
  var clientSecret = credentials.installed.client_secret;
  var clientId = credentials.installed.client_id;
  var redirectUrl = credentials.installed.redirect_uris[0];
  // client_secret.json 파일에서 시크릿키, 클라이언트 아이디, 리다이렉트 URI를 받습니다.
  // 웹 어플리케이션으로 만들었을 때에는 &#x27;installed&#x27; -&gt; &#x27;web&#x27; 으로 수정합니다.
  // ex &#x27;var clientSecret = credentials.web.client_secret;&#x27;

  var auth = new googleAuth();
  var oauth2Client = new auth.OAuth2(clientId, clientSecret, redirectUrl);

  fs.readFile(TOKEN_PATH, function(err, token) {
    // 앞서 지정한 디렉토리와 파일명으로 토큰 파일을 읽습니다.
    if (err) {
      getNewToken(oauth2Client, callback); // 토큰이 없다면 만듭니다.
    } else {
      oauth2Client.credentials = JSON.parse(token); // 토큰이 있다면 정보를 객체에 담습니다.
      callback(oauth2Client);
    }
  });
}

function getNewToken(oauth2Client, callback) {
  var authUrl = oauth2Client.generateAuthUrl({
    access_type: &#x27;offline&#x27;,
    scope: SCOPES
  });

  // 브라우저에 &#x27;authUrl&#x27;의 링크로 접근하여 데이터 접근에 동의한 후 리다이렉트 되는 URL의
  // &#x27;code=&#x27; 다음 부분을 복사합니다.
  console.log(&#x27;Authorize this app by visiting this url: &#x27;, authUrl);
  var rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout
  });
  // 복사한 &#x27;code=&#x27; 다음 부분을 입력합니다.
  // 로컬에서 테스트 할때에는 code= 다음 부분의 값이 URL 인코딩이 되어 있을 수 있으니
  // URL 디코딩한 후에 입력합니다.
  rl.question(&#x27;Enter the code from that page here: &#x27;, function(code) {
    rl.close();
    oauth2Client.getToken(code, function(err, token) {
      if (err) {
        console.log(&#x27;Error while trying to retrieve access token&#x27;, err);
        return;
      }
      oauth2Client.credentials = token;
      storeToken(token);
      callback(oauth2Client);
    });
  });
}

function storeToken(token) {
  try {
    fs.mkdirSync(TOKEN_DIR);
  } catch (err) {
    if (err.code != &#x27;EEXIST&#x27;) {
      throw err;
    }
  }
  fs.writeFile(TOKEN_PATH, JSON.stringify(token)); // 토큰을 만듭니다.
  console.log(&#x27;Token stored to &#x27; + TOKEN_PATH);
}

function listEvents(auth) {
  // 토큰 인증이 완료되면 최종적으로 실행되는 함수 입니다.
  // 아래의 예제는 구글 캘린더에 등록된 이벤트 10개의 정보를 출력합니다.
  var calendar = google.calendar(&#x27;v3&#x27;);

  calendar.events.list({
    auth: auth,
    calendarId: &#x27;primary&#x27;, // 이곳에 이벤트를 가져올 캘린더 id를 입력해야 합니다.
    timeMin: (new Date()).toISOString(),
    maxResults: 10,
    singleEvents: true,
    orderBy: &#x27;startTime&#x27;
  }, function(err, response) {
    if (err) {
      console.log(&#x27;The API returned an error: &#x27; + err);
      return;
    }
    var events = response.items;
    if (events.length == 0) {
      console.log(&#x27;No upcoming events found.&#x27;);
    } else {
      console.log(&#x27;Upcoming 10 events:&#x27;);
      for (var i = 0; i &lt; events.length; i++) {
        var event = events[i];
        var start = event.start.dateTime || event.start.date;
        console.log(&#x27;%s - %s&#x27;, start, event.summary);
      }
    }
  });


  // 다음은 구글 캘린더에 &#x27;CHEOLGUSO&#x27; 라는 이름의 캘린더가 있는지 확인하여
  // 없다면 만들고 있다면 삭제해보겠습니다.
  calendar.calendarList.list({
    auth : auth
  }, function(err, calendarList){
    if(err) console.log(err);

    var deleteCalendarId; // 캘린더 삭제를 위한 아이디 변수
    var isCGSCalendar = false; // &#x27;CHEOLGUSO&#x27; 캘린더가 존재하는지 확인하기 위한 변수

    // &#x27;CHEOLGUSO&#x27; 캘린더가 있는지 확인
    for(var i=0; i&lt;calendarList.items.length; i++){
      if(calendarList.items[i].summary === &#x27;CHEOLGUSO&#x27;){
        isCGSCalendar = true;
        deleteCalendarId = calendarList.items[i].id;
      }
    }

    if(isCGSCalendar){

      // &#x27;CHEOLGUSO&#x27; 캘린더가 있다면 캘린더를 지움
      calendar.calendars.delete({
        auth : auth,
        calendarId : deleteCalendarId
      }, function(err, calendars){
        if(err) console.log(err);
      });

    }else{

      // &#x27;CHEOLGUSO&#x27; 캘린더가 없다면 캘린더를 만듬
      calendar.calendars.insert({
        auth : auth,
        resource : {
          summary : &#x27;CHEOLGUSO&#x27;
        }
      }, function(err, calendars){
        if(err) console.log(err);
      });

    }

  });

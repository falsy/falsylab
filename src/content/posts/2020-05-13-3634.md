<h3>1. UDP Server - Node</h3>
<p class="basic-text-box">우선은 그나마 가장 익숙한 Nodejs를 이용해서 아주 간단하게 UDP 서버를 만들어보면 아래와 같답니다.</p>

<h3>2. UDP Server - C</h3>
<p class="basic-text-box">우선 관련해서 회사 동생에게 물어봤는데, 회사 동생이 윤성우님의 "열혈 TCP/IP 소켓 프로그래밍" 책을 빌려줘서 아래의 코드는 해당 책에 나오는 예제를 참고해서 약간 수정한 코드입니다.</p>
<pre class="legacy-code"><code class="language-c">// Server
#include &lt;stdio.h&gt;
#include &lt;stdlib.h&gt;
#include &lt;string.h&gt;
#include &lt;unistd.h&gt;
#include &lt;arpa/inet.h&gt;
#include &lt;sys/socket.h&gt;

void error_handling(char *message);

int main(int argc, char *argv[]) {
  int serv_sock;
  char message[1024] = {0};
  socklen_t clnt_adr_sz;
  int str_len;

  struct sockaddr_in serv_adr, clnt_adr;

  if(argc != 2) {
    printf(&quot;usage: %s &lt;port&gt;\n&quot;, argv[0]);
    exit(1);
  }

  serv_sock = socket(PF_INET, SOCK_DGRAM, 0);
  if(serv_sock == -1) {
    char err_message[] = &quot;UDP socket creation error&quot;;
    error_handling(err_message);
  }

  memset(&amp;serv_adr, 0, sizeof(serv_adr));
  serv_adr.sin_family = AF_INET;
  serv_adr.sin_addr.s_addr = htonl(INADDR_ANY);
  serv_adr.sin_port = htons(atoi(argv[1]));

  if(bind(serv_sock, (struct sockaddr*)&amp;serv_adr, sizeof(serv_adr)) == -1) {
    char err_message[] = &quot;bind() error&quot;;
    error_handling(err_message);
  }

  while(1) {
    clnt_adr_sz = sizeof(clnt_adr);
    str_len = recvfrom(serv_sock, message, 1024, 0, (struct sockaddr*)&amp;clnt_adr, &amp;clnt_adr_sz);
    printf(&quot;%s&quot;, message);
    sendto(serv_sock, message, str_len, 0, (struct sockaddr*)&amp;clnt_adr, clnt_adr_sz);
  }
  close(serv_sock);
  return 0;
}

void error_handling(char *message) {
  fputs(message, stderr);
  exit(1);
}</code></pre>

<h3>3. TCP Server - C</h3>
<p class="basic-text-box">그리고 비교를 위해 TCP 서버도 만들어봤는데요.<br>TCP는 아래의 글의 예제를 참고해서 약간 수정한 코드랍니다.<br>참고 링크. <a href="https://medium.com/from-the-scratch/http-server-what-do-you-need-to-know-to-build-a-simple-http-server-from-scratch-d1ef8945e4fa" target="_blank" rel="noopener noreferrer">https://medium.com/from-the-scratch/http-server-what-do-you-need-to-know-to-build-a-simple-http-server-from-scratch-d1ef8945e4fa</a></p>
<pre class="legacy-code"><code class="language-c">// Server
#include &lt;stdio.h&gt;
#include &lt;sys/socket.h&gt;
#include &lt;unistd.h&gt;
#include &lt;stdlib.h&gt;
#include &lt;netinet/in.h&gt;
#include &lt;string.h&gt;

#define PORT 6666

int main(int argc, char const *argv[]) {
  int server_fd, new_socket; long valread;
  struct sockaddr_in address;
  int addrlen = sizeof(address);
  char buffer[1024] = {0};

  if ((server_fd = socket(AF_INET, SOCK_STREAM, 0)) == 0) {
    perror(&quot;In socket&quot;);
    exit(EXIT_FAILURE);
  }

  address.sin_family = AF_INET;
  address.sin_addr.s_addr = INADDR_ANY;
  address.sin_port = htons( PORT );

  memset(address.sin_zero, &#x27;\0&#x27;, sizeof address.sin_zero);

  if (bind(server_fd, (struct sockaddr *)&amp;address, sizeof(address)) &lt; 0) {
    perror(&quot;In bind&quot;);
    exit(EXIT_FAILURE);
  }

  if (listen(server_fd, 10) &lt; 0) {
    perror(&quot;In listen&quot;);
    exit(EXIT_FAILURE);
  }

  while(1) {
    if ((new_socket = accept(server_fd, (struct sockaddr *)&amp;address, (socklen_t*)&amp;addrlen)) &lt; 0) {
      perror(&quot;In accept&quot;);
      exit(EXIT_FAILURE);
    }
    valread = read(new_socket, buffer, 1024);
    printf(&quot;%s&quot;, buffer);
    write(new_socket, buffer, strlen(buffer));
  }
  return 0;
  close(new_socket);
}</code></pre>

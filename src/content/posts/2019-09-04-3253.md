<p style="margin-bottom: 30px;">오늘은 간단하게 타입스크립트 + 리액트 + 웹팩 구성을 만들어 보려 합니다.</p>
<p class="left-line-box">예제 코드는 깃헙 저장소에 올려놓았습니다.
<a href="https://github.com/falsy/blog-post-example/tree/master/typescript-react-webpack-starter-kit" target="_blank" rel="noopener noreferrer">https://github.com/falsy/blog-post-example/tree/master/typescript-react-webpack-starter-kit</a>
<h3>1. 패키지 설치하기</h3>
<p class="basic-text-box">하나하나 설치하려면 힘드니까, 아래와 같은 package.json 파일을 만들고 한번에 install 할게요.</p>

<pre class="legacy-code"><code class="language-javascript">//package.json
{
  &quot;name&quot;: &quot;typescript-react-webpack-starter-kit&quot;,
  &quot;version&quot;: &quot;1.0.0&quot;,
  &quot;description&quot;: &quot;&quot;,
  &quot;main&quot;: &quot;index.js&quot;,
  &quot;scripts&quot;: {
    &quot;start&quot;: &quot;webpack-dev-server --mode development&quot;,
    &quot;build&quot;: &quot;webpack --mode development&quot;
  },
  &quot;author&quot;: &quot;falsy&quot;,
  &quot;license&quot;: &quot;WTFPL&quot;,
  &quot;devDependencies&quot;: {
    &quot;@types/react&quot;: &quot;^16.9.2&quot;, // 타입스크립트로 react를 사용하기 위한 패키지
    &quot;@types/react-dom&quot;: &quot;^16.9.0&quot;, // 타입스크립트로 react-dom을 사용하기 위한 패키지
    &quot;html-webpack-plugin&quot;: &quot;^3.2.0&quot;, // html에 번들을 넣어주는 웹팩 플러그인
    &quot;react&quot;: &quot;^16.9.0&quot;, // 리엑트
    &quot;react-dom&quot;: &quot;^16.9.0&quot;, // 리엑트 돔
    &quot;source-map-loader&quot;: &quot;^0.2.4&quot;, // 소스맵 로더
    &quot;ts-loader&quot;: &quot;^6.0.4&quot;, // 타입스크립트 로더
    &quot;typescript&quot;: &quot;^3.6.2&quot;, // 타입스크립트
    &quot;webpack&quot;: &quot;^4.30.0&quot;, // 웹팩
    &quot;webpack-cli&quot;: &quot;^3.3.1&quot;, // 웹팩 CLI
    &quot;webpack-dev-server&quot;: &quot;^3.3.1&quot; // 웹팩 개발 서버
  },
  &quot;dependencies&quot;: {}
}</code></pre>
<p class="basic-text-box">그리고 설치</p>

<pre class="legacy-code"><code class="language-default">$ npm install</code></pre>
<p class="basic-text-box">우선 최종적으로 파일 구조는 아래와 같이 할꺼에요.</p>

<pre class="legacy-code"><code class="language-default">/node_modules
  ...
/src
  App.tsx
  index.html
  index.tsx
tsconfig.json
webpack.config.js</code></pre>
<h3>2. 웹팩 설정 파일 만들기</h3>
<pre class="legacy-code"><code class="language-default">// webpack.config.js
const HTMLWeebPackPlugin = require(&#x27;html-webpack-plugin&#x27;); // 아까 설치한 플러그인이죠? html 문서에 자동으로 번들파일을 추가해줍니다.
const path = require(&#x27;path&#x27;);

module.exports = {
  entry: &#x27;./src/index.tsx&#x27;, // 처음 시작할 파일을 지정해줍니다. 지정하지 않으면 &#x27;./src/index.js&#x27;가 기본 값이기 때문에 적어줘야 해요 
  module: {
    rules: [
      {
        test: /\.tsx?$/, // .tsx 확장자로 끝나는 파일들을
        use: &#x27;ts-loader&#x27;, // ts-loader 가 트랜스파일 해줍니다.
        exclude: /node_modules/ // node_modules 디렉토리에 있는 파일들이 제외하고
      }, {
        enforce: &quot;pre&quot;,
        test: /\.js$/,
        loader: &quot;source-map-loader&quot;
      }
    ]
  },
  resolve: {
    extensions: [ &#x27;.tsx&#x27;, &#x27;.ts&#x27;, &#x27;.js&#x27; ]
  },
  output: {
    filename: &#x27;bundle.js&#x27;, // build시 만들어질 파일 번들 파일 이름
    path: path.resolve(__dirname, &#x27;dist&#x27;) // 그리고 경로 입니다.
  },
  plugins: [
    new HTMLWeebPackPlugin({
      template: &#x27;./src/index.html&#x27;,
      filename: &#x27;./index.html&#x27;
    }) // &#x27;./src/index.html&#x27; 경로의 html 파일에 번들 파일을 넣어줍니다.
  ]
};</code></pre>
<h3>3. 타입스크립트 설정 파일 만들기</h3>
<pre class="legacy-code"><code class="language-javascript">// tsconfig.json
{
  &quot;compilerOptions&quot;: {
    &quot;outDir&quot;: &quot;./dist/&quot;, // 아웃풋 될 경로
    &quot;noImplicitAny&quot;: true, // 음.. 암시적 any 타입의 표현식 및 선언에서 오류를 발생시킬지?
    &quot;module&quot;: &quot;es6&quot;,
    &quot;target&quot;: &quot;es5&quot;,
    &quot;jsx&quot;: &quot;react&quot;,
    &quot;allowJs&quot;: true // .js 파일도 함께 컴파일 할지 여부
  }
}</code></pre>

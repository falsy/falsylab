import React, {Component} from &#x27;react&#x27;;
import {DatePickerIOS, AlertIOS, Animated, Easing, StyleSheet, Text, View} from &#x27;react-native&#x27;;
// 딱 보시면 아시겠지만
// 날짜를 선택할 수 있는 네이티브 컴포넌트 : DatePickerIOS
// 알림창 네이티브 컴포넌트 : AlertIOS
// 애니메이션을 사용할 수 있는 컴포넌트 : Animated
// 애니메이션의 easing 값을 줄 수 있는 : Easing
import DashboardFooter from &#x27;../components/DashboardFooter&#x27;;
import DashboardHeader from &#x27;../components/DashboardHeader&#x27;;
import axios from &#x27;axios&#x27;;
// 저는 비동기 통신에서 주로 axios를 많이 사용해서 axios를 예로 사용했습니다.

class Dashboard extends Component {

  constructor(props) {
    super(props);
    this.state = {
      data: &#x27;&#x27;,
      date: new Date(),
      datePickerShow: false,
      left: new Animated.Value(0)
      // 애니메이션 값을 이렇게 Animated 객체로 기본값을 설정해 주셔야 합니다.
    };
  }

  componentDidMount() {
    axios.get(&#x27;http://localhost:44444/test&#x27;).then(data =&gt; {
      this.setState({
        data: data.data
      });
    });
    // 이건 이렇게 그냥 하면 오류가 나겠죠? 저는 개인적으로 간단하게 노드로 서버를 하나 띄어서 해봤어요.
    // 흔한 예로 setTimeout()을 써서 해보셔도 될 거 같아요.
    // setTimeout(() =&gt; {
    //   this.setState({
    //     data: &#x27;test&#x27;
    //   });
    // }, 500);
  }

  actionAnimate() {
    Animated.timing(
      this.state.left,
      {
        toValue: 100,
        // 애니메이션 변화 할 마지막 값?
        easing: Easing.quad,
        // 이진값
        duration: 5000,
        // 애니메이션할 시간??
      }
    ).start();
  }

  actionAlert() {
    AlertIOS.alert(
     &#x27;alert&#x27;,
     &#x27;Hello World.&#x27;
    );
    // 알림창 첫번째 파라미터는 타이틀
    // 두번째 파라미터는 메세지
    // 생략했지만 세번째 파라미터는 콜백
  }

  actionDatePicker() {
    this.setState({
      datePickerShow: true
    });
    // 이건 그냥 구현을 위해서 actionDatePicker() 가 실행되면
    // 데이트피커가 출력되도록 합니다.
  }

  render() {
    return (
      &lt;View style={{flex: 1}}&gt;
        &lt;DashboardHeader /&gt;
        &lt;View style={styles.content}&gt;
        &lt;Text style={styles.welcome}&gt;{this.state.data}&lt;/Text&gt;
          &lt;Text style={styles.welcome} onPress={this.actionAnimate.bind(this)}&gt;Animate&lt;/Text&gt;
          &lt;Text style={styles.welcome} onPress={this.actionAlert.bind(this)}&gt;Alert&lt;/Text&gt;
          &lt;Text style={styles.welcome} onPress={this.actionDatePicker.bind(this)}&gt;DatePicker&lt;/Text&gt;
        &lt;/View&gt;
        { this.state.datePickerShow ?
          &lt;View&gt;
            &lt;DatePickerIOS date={this.state.date} /&gt;
          &lt;/View&gt;
        : null }
        &lt;Animated.View style={{
          position: &#x27;absolute&#x27;,
          zIndex: 1,
          top: 100,
          width: 100,
          height: 50,
          backgroundColor: &#x27;red&#x27;,
          left: this.state.left.interpolate({
            inputRange: [0, 100],
            outputRange: [&#x27;0%&#x27;, &#x27;100%&#x27;],
          }),
          // 요게 조금 추가된 부분이랄까요?
          // 위 예에서는 변화할 값이 % 이다보니 이게 그냥 하면 안되거든요, (&#x27;100%&#x27; 는 사실 그냥 문자열이다보니??)
          // 그래서 이렇게 inputRange, outputRange 속성으로 사용할 수 있답니다.
        }}&gt;
          &lt;Text&gt;Animated&lt;/Text&gt;
        &lt;/Animated.View&gt;
        &lt;DashboardFooter /&gt;
      &lt;/View&gt;
    );
  }
}

const styles = StyleSheet.create({
  wrap: {
    flex: 1
  },
  content: {
    flex: 1,
    justifyContent: &#x27;center&#x27;,
    alignItems: &#x27;center&#x27;,
    backgroundColor: &#x27;#fff&#x27;,
  },
  welcome: {
    fontSize: 20,
    textAlign: &#x27;center&#x27;,
    color: &#x27;#000&#x27;
  }
});

export default Dashboard</code></pre>

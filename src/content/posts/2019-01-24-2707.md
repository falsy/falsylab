<p style="margin-bottom: 30px;">어제인가, 그제인가 우연히 회사에서 코딩 테스트 문제를 풀어 볼 일이 있었습니다. 총 4문제였는데, 어디서 온 자신감인지 당연히 다 풀 수 있을 줄 알았는데, 마지막 문제를 한 시간 동안 풀지 못했습니다.</p>
<p class="basic-text-box">하...</p>
<p class="left-line-box">그 문제는 이러했습니다.</p>
<p class="basic-text-box">정확하게 기억나지 않지만 함수에 인자 값으로 x, y를 주며 x는 행 y는 열을 나타낸다. 그리고 1부터 시작해서 시계방향 소용돌이 모양으로 값을 출력한다.
예를 들어 x=2, y=2가 주어진다면 출력값은 아래와 같습니다.
<pre class="legacy-code"><code class="language-javascript">1  2
4  3</code></pre>
<p class="basic-text-box">조금 더 이해를 위해 x=5, y=7가 주어지면 출력해야 하는 값은 아래와 같습니다.</p>

<pre class="legacy-code"><code class="language-javascript">1  2  3  4  5  6  7
20 21 22 23 24 25 8
19 32 33 34 35 26 9
18 31 30 29 28 27 10
17 16 15 14 13 12 11</code></pre>
<p class="basic-text-box">약간의 오기가 생겨, 오늘은 이 문제를 풀어보았습니다.</p>

<h3>풀이</h3>
<pre class="legacy-code"><code class="language-javascript">function main(x, y) {

	function directionFnc() {
		let direction = [&#x27;right&#x27;, &#x27;bottom&#x27;, &#x27;left&#x27;, &#x27;top&#x27;];
		let idx = -1;
		return () =&gt; {
			idx = idx + 1 &gt; 3 ? 0 : idx + 1;
			return direction[idx];
		};
	}

	const direction = directionFnc();
	const dataCase = [];
	
	let nowDirection = direction();
	let endCheck = false;
	let no = 1;
	let posX = 0;
	let posY = 0;

	for(let i=0; i&lt;x; i++) {
		dataCase.push([]);
		for(let j=0; j&lt;y; j++) {
			dataCase[i].push(0);
		}
	}

	function moveFillData() {

		if(typeof dataCase[posX] === &#x27;undefined&#x27; || typeof dataCase[posX][posY] === &#x27;undefined&#x27;) return dataCase;
		
		switch(nowDirection) {
			case &#x27;right&#x27;:
				if(posY + 1 === y || dataCase[posX][posY + 1] !== 0) {
					if(endCheck &amp;&amp; dataCase[posX + 1][posY] !== 0 &amp;&amp; dataCase[posX][posY] !== 0) return dataCase;
					nowDirection = direction();
					endCheck = true;
					dataCase[posX][posY] = no;
					no += 1;
					posX += 1;
				} else {
					endCheck = false;
					dataCase[posX][posY] = no;
					no += 1;
					posY += 1;
				}
				return moveFillData();
			case &#x27;bottom&#x27;:
				if(posX + 1 === x || dataCase[posX + 1][posY] !== 0) {
					if(endCheck &amp;&amp; dataCase[posX][posY - 1] !== 0 &amp;&amp; dataCase[posX][posY] !== 0) return dataCase;
					nowDirection = direction();
					endCheck = true;
					dataCase[posX][posY] = no;
					no += 1;
					posY -= 1;
				} else {
					endCheck = false;
					dataCase[posX][posY] = no;
					no += 1;
					posX += 1;
				}
				return moveFillData();
			case &#x27;left&#x27;:
				if(posY - 1  === -1 || dataCase[posX][posY - 1] !== 0) {
					if(endCheck &amp;&amp; dataCase[posX - 1][posY] !== 0 &amp;&amp; dataCase[posX][posY] !== 0) return dataCase;
					nowDirection = direction();
					endCheck = true;
					dataCase[posX][posY] = no;
					no += 1;
					posX -= 1;
				} else {
					endCheck = false;
					dataCase[posX][posY] = no;
					no += 1;
					posY -= 1;
				}
				return moveFillData();
			case &#x27;top&#x27;:
				if(posX - 1  === -1 || dataCase[posX - 1][posY] !== 0) {
					if(endCheck &amp;&amp; dataCase[posX][posY + 1] !== 0 &amp;&amp; dataCase[posX][posY] !== 0) return dataCase;
					nowDirection = direction();
					endCheck = true;
					dataCase[posX][posY] = no;
					no += 1;
					posY += 1;
				} else {
					endCheck = false;
					dataCase[posX][posY] = no;
					no += 1;
					posX -= 1;
				}
				return moveFillData();
		}
	};
	
	return moveFillData();
}

main(3, 2);
// 1 2
// 6 3
// 5 4</code></pre>

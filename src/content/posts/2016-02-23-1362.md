<div id="review_text">
<p class="basic-text-box">
오늘은 자바스크립트 정규표현식에 대해 포스팅을 해보려 합니다. 너무 복잡하고 어렵지 않은 가벼운 첫걸음을 위한 글입니다.
</p>
 
<pre class="legacy-code"><code class="language-javascript">// 이메일 검증
var emailCheck = /^(([^&amp;amp;lt;&amp;amp;gt;()[\]\\.,;:\s@\&quot;]+(\.[^&amp;amp;lt;&amp;amp;gt;()[\]\\.,;:\s@\&quot;]+)*)|(\&quot;.+\&quot;))@((\[[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\])|(([a-zA-Z\-0-9]+\.)+[a-zA-Z]{2,}))$/;</code></pre> 

<p class="left-line-box">보통 처음로 접하는 정규표현식이 이렇게 생겨서 더 어렵게 느껴지는게 아닐까요..</p>

<p class="basic-text-box">우선 정규표현식은 다양하게 사용이 되지만 몇가지 예로는 
.test(), .match(), .search(), .replace() 등.. 이 있습니다.
.test() - 해당 문자열이 있는지 true, false 출력
.match() - 해당 문자열이 있다면 그 문자열을 출력
.search() - 해당 문자열이 몇번째에 있는지 출력
.replace() - 해당 문자열을 다른 문자로 바꿉니다.</p>

<h3>1. .test()</h3>

 
<pre class="legacy-code"><code class="language-javascript">var text = &#x27;abc def ghi jkl abc def&#x27;; // 대상 문자열
var testReg = /abc/; // 찾을 문자열
var result = testReg.test(text); 

console.log(result); // true</code></pre> 


<h3>2. .match()</h3> 

 
<pre class="legacy-code"><code class="language-javascript">var text = &#x27;abc def ghi jkl abc def&#x27;; // 대상 문자열
var matchReg = /abc/; // 찾을 문자열
var result = text.match(matchReg); 

console.log(result); // [&quot;abc&quot;, index: 0, input: &quot;abc def ghi jkl abc def&quot;]</code></pre> 


<h3>3. .search()</h3> 

 
<pre class="legacy-code"><code class="language-javascript">var text = &#x27;abc def ghi jkl abc def&#x27;; // 대상 문자열
var searchReg = /def/; // 찾을 문자열
var result = text.search(searchReg); 

console.log(result); // 4</code></pre> 


<h3>4. .replace()</h3> 

 
<pre class="legacy-code"><code class="language-javascript">var text = &#x27;abc def ghi jkl abc def&#x27;; // 대상 문자열
var replaceReg = /def/; // 찾을 문자열
var result = text.replace(replaceReg, &#x27;가나다&#x27;); 

console.log(result); // &#x27;abc 가나다 ghi jkl abc def&#x27;</code></pre> 



<p class="basic-text-box">위의 예제들은 아주아주 간단한 소개로 자세한 정보는 '모질라 개발자 네트워크(<a href="https://developer.mozilla.org/ko/" target="_blank">https://developer.mozilla.org/ko/</a>)' 같은 곳들을 참고하시면 좋을 것 같습니다.</p>

<h3>정규 표현식을 시작하기 전에</h3>
 
<p class="basic-text-box">
일단 기본적으로 정규식의 문자열은 '/'와 '/' 사이에 적습니다. 그리고 각각의 옵션 들이 특수문자로 사용되기 때문에 문자열로 특수문자를 사용하기 위해서는 '\'(역슬러시)를 앞에 적어서 사용할 수 있습니다.
<span class="bg-point-text">[ex. var regText = /abc\=abc/;]</span>
</p>

<h3>정규 표현식 시작</h3>

 
<pre class="legacy-code"><code class="language-javascript">var text = &#x27;&lt;ul id=&quot;aaa&quot;&gt;&lt;li&gt;Hello &lt;span id=&quot;aaa&quot;&gt;World&lt;/span&gt;&lt;/li&gt;&lt;/ul&gt;&#x27;; // 대상 문자열

// 위 문자열의 ul와 li엘리먼트를 div로 바꾸고 aaa로 반복 사용된 id 프로퍼티를 class로 바꿔보려합니다.
var replaceReg = /(&lt;\/?(ul|li))|((id\=)+)/gi;

// 마지막의 /gi 에서
// g 는 대상 문자열 모든 곳에서 찾습니다. g가 없다면 첫번째 찾은 것에서 찾는 것을 멈춥니다.
// i 는 대소문자를 가리지 않고 찾습니다.
// 우선 가장 큰 영역부터 보게되면 /( ... | ... )/
// 여기에서 |는 또는, or 을 의미합니다. 둘중 하나라도 있을때?
// 첫번째 ( ... ) 를 보자면
// 첫번째로 &#x27;&lt;&#x27; 를 가지고 있으면서
// ? 앞의 &#x27;/&#x27;는 있을 수도 있는 것도 포함하고 없는 것도 포함한다.
// 다시 (ul|li) ul 또는 li
// 해석하자면 &lt;ul &lt;/ul &lt;li &lt;/li 를 찾을 것입니다.
// 두번째의 ( ... ) 는
// + 앞의 (id=)가 1나 이상 포함하고 있다면
// 이번 예제에서는 꼭 +를 하지 않아도 상관없겠네요.
// /(&lt;\/?(ul|li))|((id\=))/gi

text = text.replace(replaceReg, function(x){

    //.replace() 의 두번째 인자는 함수로 사용할 수 있고 x에는 정규표현식에 해당된 문자가 됩니다.
    if(x.search(&#x27;&lt;&#x27;) === -1) x = &#x27;class=&#x27;; // 대상이 &lt; 를 가지고 있지 않다면 (x는 &#x27;id=&#x27; 만 해당이 됩니다.)
    else if(x.search(&#x27;/&#x27;) === -1) x = &#x27;&lt;div&#x27;; // 대상이 / 를 가지고 있지 않다면 (x는 &#x27;&lt;ul&#x27;, &#x27;&lt;li&#x27; 가 됩니다.)
    else x = &#x27;&lt;/div&#x27;; // 나머지 &#x27;&lt;/ul&#x27;, &#x27;&lt;/li&#x27;
    return x;
});

console.log(text); // &lt;div class=&quot;aaa&quot;&gt;&lt;div&gt;Hello &lt;span class=&quot;aaa&quot;&gt;World&lt;/span&gt;&lt;/div&gt;&lt;/div&gt;</code></pre>

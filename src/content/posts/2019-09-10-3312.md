<h3>2. Webpack 설정 파일 만들기</h3>
<pre class="legacy-code"><code class="language-javascript">// webpack.config.js

  output: {
    path: resolve(__dirname, &#x27;src/dist&#x27;),
    publicPath: &#x27;/&#x27;
  },
  module: {
    rules: [
      {
        test: /\.js$/,
        exclude: /node_modules/,
        use: {
          loader: &#x27;babel-loader&#x27;
        }
      },
      {
        test: /\.html$/,
        use: [
          {
            loader: &#x27;html-loader&#x27;,
            options: { minimize: true }

        ]
      },
      {
        test: /.scss$/,

            loader: &#x27;css-loader&#x27;,
            options: {
              modules: true
            }

          &#x27;sass-loader&#x27;

    ]
  },
  plugins: [
    new HTMLWeebPackPlugin({
      template: resolve(__dirname, &#x27;src/index.html&#x27;),
      filename: &#x27;./index.html&#x27;
    })
  ]
};</code></pre>

<h3>8. 테스트 코드 작성</h3>
<pre class="legacy-code"><code class="language-javascript">// __TEST__/sample.spec.js

import Adapter from &#x27;enzyme-adapter-react-16&#x27;;
import Sample from &#x27;../components/Sample&#x27;;
Enzyme.configure({ adapter: new Adapter() });

test(&#x27;check status component redner&#x27;, () =&gt; {
  const component = shallow(&lt;Sample /&gt;);
  expect(component.text()).toEqual(&#x27;SAMPLE PAGE&#x27;);
});</code></pre>

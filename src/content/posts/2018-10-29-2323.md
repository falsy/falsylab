<pre class="legacy-code"><code class="language-javascript">// line-chart.js
const width = 500;
const height = 500;
const margin = {top: 40, right: 40, bottom: 40, left: 40};
const data = [
  {date: new Date(&#x27;2018-01-01&#x27;), value: 10},
  {date: new Date(&#x27;2018-01-02&#x27;), value: 20},
  {date: new Date(&#x27;2018-01-03&#x27;), value: 30},
  {date: new Date(&#x27;2018-01-04&#x27;), value: 25},
  {date: new Date(&#x27;2018-01-05&#x27;), value: 35},
  {date: new Date(&#x27;2018-01-06&#x27;), value: 45},
  {date: new Date(&#x27;2018-01-07&#x27;), value: 60},
  {date: new Date(&#x27;2018-01-08&#x27;), value: 50}
];

const x = d3.scaleTime()
  .domain(d3.extent(data, d =&gt; d.date))
  .range([margin.left, width - margin.right]);

const y = d3.scaleLinear()
  .domain([0, d3.max(data, d =&gt; d.value)]).nice()
  .range([height - margin.bottom, margin.top]);

const xAxis = g =&gt; g
  .attr(&quot;transform&quot;, `translate(0,${height - margin.bottom})`)
  .call(d3.axisBottom(x).ticks(width / 90).tickSizeOuter(0));

const yAxis = g =&gt; g
  .attr(&quot;transform&quot;, `translate(${margin.left},0)`)
  .call(d3.axisLeft(y))
  .call(g =&gt; g.select(&quot;.domain&quot;).remove())
  .call(g =&gt; {
    return g.select(&quot;.tick:last-of-type text&quot;).clone()
      .attr(&quot;x&quot;, 3)
      .attr(&quot;text-anchor&quot;, &quot;start&quot;)
      .attr(&quot;font-weight&quot;, &quot;bold&quot;)
      .attr(&quot;font-size&quot;, &#x27;20px&#x27;)
      .text(&#x27;Y축&#x27;)
    });

const line = d3.line()
  .defined(d =&gt; !isNaN(d.value))
  .x(d =&gt; x(d.date))
  .y(d =&gt; y(d.value));

const svg = d3.select(&#x27;body&#x27;).append(&#x27;svg&#x27;).style(&#x27;width&#x27;, width).style(&#x27;height&#x27;, height);
svg.append(&quot;path&quot;)
  .datum(data)
  .attr(&quot;fill&quot;, &quot;none&quot;)
  .attr(&quot;stroke&quot;, &quot;steelblue&quot;)
  .attr(&quot;stroke-width&quot;, 1)
  .attr(&quot;stroke-linejoin&quot;, &quot;round&quot;)
  .attr(&quot;stroke-linecap&quot;, &quot;round&quot;)
  .attr(&quot;d&quot;, line);
svg.append(&#x27;g&#x27;).call(xAxis);
svg.append(&#x27;g&#x27;).call(yAxis);
svg.node();</code></pre>

<pre class="legacy-code"><code class="language-javascript">// x 축으로 사용될 값을 설정합니다.

  // 내부 소스를 보지 않아서 확실하진 않지만 X축으로 사용할 값을 직접 원시 값을 넣어주지 않아도 위 메서드를 통해서 알아서 해주는 것 같습니다.

  // .range()는 위치값으로 최소값의 위치와 최대값의 위치를 배열로 받아 위 .domain의 값과 매칭하는 듯 합니다.

const y = d3.scaleLinear()
  // .scaleLinear()는 위 .scaleTime() 과 비슷한 역할을 하지만 얜 선의 범위로 설정? 선언? 해준 것 입니다.

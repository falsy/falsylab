<h3>2. 구글맵 자바스크립트 API</h3>
<p class="highlight_color2" style="margin-bottom: -10px;">2-1. Javascript API</p>

<pre class="legacy-code"><code class="language-javascript">jQuery(document).ready(function(){
    google.maps.event.addDomListener(window, &#x27;load&#x27;, initialize);

function initialize(){
    if($(&quot;#google_map&quot;).length) {

    var mapOptions = { //구글 맵 옵션 설정
        zoom : 16, //기본 확대율
        center : new google.maps.LatLng(37.5651, 126.98955), // 지도 중앙 위치
        scrollwheel : false, //마우스 휠로 확대 축소 사용 여부
        mapTypeControl : false //맵 타입 컨트롤 사용 여부
    };

    var map = new google.maps.Map(document.getElementById(&#x27;google_map&#x27;), mapOptions); //구글 맵을 사용할 타겟
    
    var image = &#x27;https://cheolguso.com/img/iconimg.png&#x27;; //마커 이미지 설정

    var marker = new google.maps.Marker({ //마커 설정
        map : map,
        position : map.getCenter(), //마커 위치
        icon : image //마커 이미지
    });
    google.maps.event.addDomListener(window, &quot;resize&quot;, function() { //리사이즈에 따른 마커 위치
        var center = map.getCenter();
        google.maps.event.trigger(map, &quot;resize&quot;);
        map.setCenter(center); 
    });

    }//if-end
}//function-end

});//jQuery-end</code></pre>

<h3>5. 구글맵 마커 추가</h3>
<p class="highlight_color2" style="margin-bottom: -10px;">구글맵에 2개의 마커 사용하기</p>

<pre class="legacy-code"><code class="language-javascript">google.maps.event.addDomListener(window, &#x27;load&#x27;, initialize);

function initialize(){
    var mapOptions = {
        zoom : 16,
        center : new google.maps.LatLng(37.544553, 127.017309), //기준이 될 가운데 좌표
        disableDefaultUI : false,
        scrollwheel : false,
        mapTypeControl : false,
        zoomControl : false,
        zoomControlOptions : false,
        streetViewControl : false,
        panControl :false
    };
 
    var map = new google.maps.Map(document.getElementById(&#x27;google_map&#x27;), mapOptions);
    
    google.maps.event.addDomListener(window, &quot;resize&quot;, function() {
        var center = map.getCenter();
        google.maps.event.trigger(map, &quot;resize&quot;);
        map.setCenter(center); 
    });

    var company = [
	[&#x27;company 1&#x27;, 37.5375772, 127.0062798, 2], // 1번 타이틀, 마커 좌표값, 우선순위(z-index)
	[&#x27;company 2&#x27;, 37.544553, 127.017309, 1] // 2번 타이틀, 마커 좌표값, 우선순위(z-index)
	];

    setMarkers(map, company);
}

function setMarkers(map, locations){
	for(var i=0; i &lt; locations.length; i++){
		
		if(i == 0) var image = &#x27;https://cheolguso.com/img/iconimg.png&#x27;; // 1번 마커 이미지
		if(i == 1) var image = &#x27;https://cheolguso.com/img/iconimg2.png&#x27;; // 2번 마커 이미지
		
		var compa = locations[i];
		var myLatLng = new google.maps.LatLng(compa[1], compa[2]);
		var marker = new google.maps.Marker({
			position: myLatLng,
			map: map,
			icon : image,
			title : compa[0],
			zIndex : compa[3]
		});
	}	
}</code></pre>

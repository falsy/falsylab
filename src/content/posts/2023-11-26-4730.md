<h3>7. useContext를 사용한 커스텀 훅 테스트하기</h3>

<pre class="legacy-code"><code class="language-javascript">// src/__test__/context.test.tsx
/**
 * @jest-environment jsdom
 */
import React, { useContext, useState } from &#x27;react&#x27;
import { act, renderHook } from &#x27;@testing-library/react&#x27;
import &#x27;@testing-library/jest-dom&#x27;

const AppContext = React.createContext(null)

const AppProvider = ({ children }) =&gt; {
  const [isBoolean, setBoolean] = useState(false)

  return (
    &lt;AppContext.Provider value={{ isBoolean, setBoolean }}&gt;{children}&lt;/AppContext.Provider&gt;
  )
}

const useAppHook = () =&gt; {
  const { isBoolean, setBoolean } = useContext(AppContext)

  return { isBoolean, setBoolean }
}

test(&#x27;useAppHook 을 사용하여 isBoolean 값을 토글할 수 있다.&#x27;, async () =&gt; {
  const wrapper = ({ children }) =&gt; &lt;AppProvider&gt;{children}&lt;/AppProvider&gt;
  const { result } = renderHook(() =&gt; useAppHook(), { wrapper })
  
  expect(result.current.isBoolean).toBe(false)

  act(() =&gt; {
    result.current.setBoolean(true)
  })

  expect(result.current.isBoolean).toBe(true)
})</code></pre>

<p class="basic-text-box">짜잔! 이렇게 renderHook 과 act 를 사용하면 useContext를 사용한 커스텀 훅을 테스트 할 수 있답니다.</p>

<h3>9. 이벤트 테스트 하기</h3>

<p class="basic-text-box">간단하게 터치 이벤트를 만들어서 테스트 해볼게요.</p>

<pre class="legacy-code"><code class="language-javascript">// src/__test__/evnet.test.tsx
/**
 * @jest-environment jsdom
 */
import React, { TouchEvent, useState } from &#x27;react&#x27;
import { fireEvent, render, screen } from &#x27;@testing-library/react&#x27;
import &#x27;@testing-library/jest-dom&#x27;

const App = () =&gt; {
  const [touchStatus, setTouchStatus] = useState(&#x27;&#x27;)

  const handleTouchStart = (e: TouchEvent&lt;HTMLDivElement&gt;) =&gt; {
    const touchY = e.touches[0].clientY
    setTouchStatus(`start clientY: ${touchY}`)
  }

  const handleTouchMove = (e) =&gt; {
    const touchY = e.touches[0].clientY
    setTouchStatus(`move clientY: ${touchY}`)
  }

  const handleTouchEnd = () =&gt; {
    setTouchStatus(&#x27;end&#x27;)
  }

  return (
    &lt;div 
      className=&#x27;event-area&#x27; 
      onTouchStart={(e) =&gt; handleTouchStart(e)}
      onTouchMove={(e) =&gt; handleTouchMove(e)}
      onTouchEnd={handleTouchEnd}
    &gt;
      hello world
      &lt;p&gt;{touchStatus}&lt;/p&gt;
    &lt;/div&gt;
  )
}

test(&#x27;터치 이벤트 상태의 변화에 따른 값을 테스트 할 수 있다.&#x27;, async () =&gt; {
  render(
    &lt;App /&gt;
  )

  const eventArea = document.querySelector(&#x27;.event-area&#x27;)

  fireEvent.touchStart(eventArea, { touches: [{ clientY: 100 }] })
  expect(screen.getByText(&#x27;start clientY: 100&#x27;)).toBeInTheDocument()

  fireEvent.touchMove(eventArea, { touches: [{ clientY: 50 }] })
  expect(screen.getByText(&#x27;move clientY: 50&#x27;)).toBeInTheDocument()
  
  fireEvent.touchEnd(eventArea)
  expect(screen.getByText(&#x27;end&#x27;)).toBeInTheDocument()
})</code></pre>

<h3>2. Actions</h3>
<p class="basic-text-box">여기서 액션은, 관례적으로 비동기 전환을 사용하는 함수를 이야기하는데요.<br>아주 간단한 상황을 예로 들어서, React v18의 구성으로 구현한다면 아래와 같이 만들어 볼 수 있는데요.</p>
<pre class="legacy-code"><code class="language-javascript">import { useState } from &quot;react&quot;

function updateName(name) {
  return new Promise((resolve) =&gt; {
    setTimeout(() =&gt; {
      resolve({
        isSuccess: true,
        name: name,
      })
    }, 1000)
  })
}

export default function React18() {
  const [displayName, setDisplayName] = useState(&quot;&quot;)
  const [name, setName] = useState(&quot;&quot;)
  const [error, setError] = useState(false)
  const [isPending, setIsPending] = useState(false)

  const handleSubmit = async () =&gt; {
    setIsPending(true)
    const { isSuccess, name: newName } = await updateName(name)
    setIsPending(false)
    if (isSuccess === false) {
      setError(true)
      return
    }
    setDisplayName(newName)
  }

  return (
    &lt;div&gt;
      &lt;p&gt;name: {displayName}&lt;/p&gt;
      &lt;input value={name} onChange={(event) =&gt; setName(event.target.value)} /&gt;
      &lt;button onClick={handleSubmit} disabled={isPending}&gt;
        Update
      &lt;/button&gt;
      {error &amp;&amp; &lt;p&gt;error&lt;/p&gt;}
    &lt;/div&gt;
  )
}</code></pre>
<p class="basic-text-box">비동기 전환을 사용하는 함수를 호출하는 것을 액션이라고 한다고 했었는데요. 위 예시에서는 "handleSubmit"가 액션 함수가 되겠죠?!</p>
<p class="basic-text-box">위 코드는 보면 useState를 사용해서 Pending 상태와 Error 상태를 확인하고 있는데요, 이제 React v19의 코드로 다시 구성해 본다면,</p>
<pre class="legacy-code"><code class="language-javascript">import { useState, useTransition } from &quot;react&quot;

function updateName(name) {
  return new Promise((resolve) =&gt; {
    setTimeout(() =&gt; {
      resolve({
        isSuccess: true,
        name: name,
      })
    }, 1000)
  })
}

export default function React19() {
  const [displayName, setDisplayName] = useState(&quot;&quot;)
  const [name, setName] = useState(&quot;&quot;)
  const [error, setError] = useState(false)
  const [isPending, startTransition] = useTransition()

  const handleSubmit = () =&gt; {
    startTransition(async () =&gt; {
      const { isSuccess, name: newName } = await updateName(name)
      if (isSuccess === false) {
        setError(true)
        return
      }
      setDisplayName(newName)
    })
  }

  return (
    &lt;div&gt;
      &lt;p&gt;name: {displayName}&lt;/p&gt;
      &lt;input value={name} onChange={(event) =&gt; setName(event.target.value)} /&gt;
      &lt;button onClick={handleSubmit} disabled={isPending}&gt;
        Update
      &lt;/button&gt;
      {error &amp;&amp; &lt;p&gt;error&lt;/p&gt;}
    &lt;/div&gt;
  )
}</code></pre>
<p class="basic-text-box">위와 같이 useTransition 훅을 사용하여 Pending 상태를 처리할 수 있답니다.</p>
<p class="left-line-box">useTransition는 React v18에서 추가된 훅이지만, v18에서 startTransition는 동기 함수만 인자로 받을 수 있었는데 v19에서 비동기 함수도 사용할 수 있도록 추가되었습니다.</p>
<h3>3. useActionState</h3>
<p class="basic-text-box">그리고 일반적인 경우의 작업을 더 쉽게 하기 위해서 useActionState 라는 함수가 추가되었는데요.<br>예를 들면 아래와 같습니다.</p>
<pre class="legacy-code"><code class="language-javascript">import { useState, useActionState } from &quot;react&quot;

function updateName(name) {
  return new Promise((resolve) =&gt; {
    setTimeout(() =&gt; {
      resolve({
        success: true,
        name: name,
      })
    }, 1000)
  })
}

export default function ActionState() {
  const [name, setName] = useState(&quot;&quot;)

  const [state, action, pending] = useActionState(
    async (previousState, newState) =&gt; {
      const res = await updateName(newState.nam)
      return res

  const handleClick = () =&gt; {
    action({ name })
  }

  return (
    &lt;div&gt;

<p class="basic-text-box">useActionState를 사용해서 구성하니 코드가 많이 간단해졌죠?! handleClick도 없어도 되지만 newState 인자의 의미를 확실히 나타내려고 일부러 추가해 보았습니다.</p>
<h3>4. form action</h3>
<p class="basic-text-box">이어서 React Dom의 form에 action이 추가되었는데요. 앞에 이야기했던 useActionState 훅을 사용해서 아래와 같이 구현할 수 있답니다.</p>
<pre class="legacy-code"><code class="language-javascript">import { useActionState } from &quot;react&quot;

function updateName(name) {

        success: true,
        name: name,
      })

export default function ActionStateForm() {

<h3>5. useFormStatus</h3>
<p class="basic-text-box">이어서 React Dom에 useFormStatus라는 훅이 추가되었는데요, 예를 들어서 form이 있고 만약 button 요소가 별도의 컴포넌트로 구성되어 있을 때 form의 pending 상태를 button 컴포넌트에서 알기 위해서는 props로 전달하거나 context를 사용해야 하는데 이를 useFormStatus를 사용하여 구성할 수 있습니다.<br>예를 들면,</p>

import FormStatusBtn from &quot;./FormStatusBtn&quot;

function updateName(name) {

<p class="basic-text-box">위와 같이 form이 있고</p>

<h3>6. useOptimistic</h3>
<p class="basic-text-box">낙관적 업데이트는 데이터를 변경하는 통신을 할 때, 변경 사항을 UI에 미리 반영하는 방법을 이야기하는 데요, 간단하게 예를 들면,</p>
<pre class="legacy-code"><code class="language-javascript">const handleSubmit = () =&gt; {
  const cache = name
  startTransition(async () =&gt; {
    setDisplayName(newName)
    try {
      await updateName(newName)
    } catch {
      setDisplayName(cache)
    }
  })
}</code></pre>
<p class="basic-text-box">위와 같이, 이름 그대로 낙관적으로 판단하고, 우선 업데이트 통신이 성공했다고 가정하고 데이터를 선 반영하고 통신에서 실패했을 때에는 캐시 해놓은 값으로 다시 값을 복구하는 업데이트 방법으로, 일반적으로도 많이 사용하는 방법인데요.</p>
<p class="basic-text-box">useOptimistic 라는 훅이 추가되어서 위와 같은 낙관적 업데이트 처리를 조금 더 간단하게 구현할 수 있게 되었습니다. 간단하게 예를 들면,</p>
<pre class="legacy-code"><code class="language-javascript">import { useOptimistic, useState, useTransition } from &quot;react&quot;

let toggle = true

function updateName(name) {
  return new Promise((resolve, reject) =&gt; {
    setTimeout(() =&gt; {
      if (toggle) {
        resolve({
          success: true,
          name: name,
        })
      } else {
        reject(&quot;Error&quot;)
      }
      toggle = !toggle
    }, 1000)
  })
}

export default function Optimistic() {
  const [originData, setOriginData] = useState(&quot;&quot;)
  const [optimisticName, setOptimisticName] = useOptimistic(originData)
  const [name, setName] = useState(originData)
  const [error, setError] = useState(false)
  const [isPending, startTransition] = useTransition()

  const handleSubmit = () =&gt; {
    setError(false)
    startTransition(async () =&gt; {
      setOptimisticName(name)
      try {
        const { name: resName } = await updateName(name)
        setOriginData(resName)
      } catch (e) {
        setError(true)
      }
    })
  }

  return (
    &lt;div&gt;
      &lt;p&gt;name: {optimisticName}&lt;/p&gt;
      &lt;input value={name} onChange={(event) =&gt; setName(event.target.value)} /&gt;
      &lt;button onClick={handleSubmit} disabled={isPending}&gt;
        Update
      &lt;/button&gt;
      {error &amp;&amp; &lt;p&gt;error&lt;/p&gt;}
    &lt;/div&gt;
  )
}</code></pre>
<p class="basic-text-box">toggle 이라는 변수를 두고 성공과 실패를 반복하도록 구성했는데요, 위와 같이 handleSubmit 이벤트가 실행되면 우선 setOptimisticName에 의해 값이 면저 변경되고 성공하면 유지되지만, 만약 통신이 실패한다면 이전의 값으로 돌아갑니다.</p>

<h3>7. use</h3>
<p class="basic-text-box">use 라는 비동기 작업을 처리를 위한, Promise를 지원하는 새로운 훅이 추가되었습니다. 기존의 useEffect와 달리 use는 컴포넌트 렌더링 시점에서 데이터를 가져오는 Promise를 기다리고 데이터가 준비될 때까지 컴포넌트가 중단됩니다. 그리고 기존의 React의 훅들은 컴포넌트의 최상위에만 호출할 수 있었는데 use는 조건부로도 호출이 가능하다는 특징이 있습니다.</p>
<p class="basic-text-box">간단하게 구현해보면 아래와 같습니다.</p>
<pre class="legacy-code"><code class="language-javascript">import { use, Suspense } from &quot;react&quot;

function getName() {
  return new Promise((resolve) =&gt; {
    setTimeout(() =&gt; {
      resolve({
        success: true,
        name: &quot;falsy&quot;,
      })
    }, 1000)
  })
}

function UserName({ userNamePromise }) {
  const user = use(userNamePromise)

  return &lt;p&gt;{user.name}&lt;/p&gt;
}

export default function UseHook() {
  const userNamePromise = getName()

  return (
    &lt;Suspense fallback={&lt;div&gt;Loading...&lt;/div&gt;}&gt;
      &lt;UserName userNamePromise={userNamePromise} /&gt;
    &lt;/Suspense&gt;
  )
}</code></pre>
<p class="basic-text-box">그리고 use의 특징 중에 하나인 조건부로 호출하는 예시는 아래와 같습니다.</p>
<pre class="legacy-code"><code class="language-javascript">import { use, Suspense } from &quot;react&quot;

function getName() {
  return new Promise((resolve) =&gt; {
    setTimeout(() =&gt; {
      resolve({
        success: true,
        name: &quot;falsy&quot;,
      })
    }, 1000)
  })
}

function UserName({ userNamePromise, useFallback }) {
  if (useFallback) {
    return &lt;p&gt;Guest&lt;/p&gt;
  }

  const user = use(userNamePromise)
  return &lt;p&gt;{user.name}&lt;/p&gt;
}

export default function UseHook() {
  const userNamePromise = getName()
  const shouldUseFallback = true

  return (
    &lt;Suspense fallback={&lt;div&gt;Loading...&lt;/div&gt;}&gt;
      &lt;UserName
        userNamePromise={userNamePromise}
        useFallback={shouldUseFallback}
      /&gt;
    &lt;/Suspense&gt;
  )
}</code></pre>

<p class="basic-text-box">

<pre class="legacy-code"><code class="language-javascript">import { useRef } from &quot;react&quot;

<p class="basic-text-box">
  이전까지는 아래와 같이 props 이름을 'ref'가 아닌 다른 이름을 사용하거나
</p>

<pre class="legacy-code"><code class="language-javascript">import { useRef } from &quot;react&quot;

function ChildRef({ TestRef }) {
  return &lt;input ref={TestRef} /&gt;
}

export default function ParentRef() {
  const TestRef = useRef(null)

  return (

export default function ParentRef() {
  const TestRef = useRef(null)

  return (
    &lt;div&gt;

<p class="left-line-box">
  이후 버전에서 'forwardRef'는 점차 사용이 중단될 예정입니다.

<h3>8-3. Context의 Provider</h3>

<p class="basic-text-box">
  생성된 Context를 그대로 Provider로 사용할 수 있습니다.
</p>

<pre class="legacy-code"><code class="language-javascript">import { createContext } from &quot;react&quot;

const NewContext = createContext()

export default function ProviderComp({ children }) {
  return (
    &lt;NewContext value={{ name: &quot;Provider&quot; }}&gt;
      {children}
    &lt;/NewContext&gt;
  )
}</code></pre>

<p class="basic-text-box">
  이전까지는 아래와 같이 'Context.Provider'를 사용해야 했습니다.
</p>

<pre class="legacy-code"><code class="language-javascript">import { createContext } from &quot;react&quot;

const NewContext = createContext()

export default function ProviderComp({ children }) {
  return (
    &lt;NewContext.Provider value={{ name: &quot;Provider&quot; }}&gt;
      {children}
    &lt;/NewContext.Provider&gt;
  )
}</code></pre>

<p class="left-line-box">
  이후 버전에서 'Context.Provider'는 점차 사용이 중단될 예정입니다.
</p>

<h3>8-6. 문서 메타데이터 지원</h3>

<p class="basic-text-box">
  &lt;title&gt;, &lt;link&gt;, &lt;meta&gt;와 같은 문서 메타데이터 태그를 네이티브로 지원합니다.
</p>

<pre class="legacy-code"><code class="language-javascript">function BlogPost({ post }) {
  return (
    &lt;article&gt;
      &lt;h1&gt;{post.title}&lt;/h1&gt;
      &lt;title&gt;{post.title}&lt;/title&gt;
      &lt;meta name=&quot;author&quot; content=&quot;Josh&quot; /&gt;
      &lt;link rel=&quot;author&quot; href=&quot;https://twitter.com/joshcstory/&quot; /&gt;
      &lt;meta name=&quot;keywords&quot; content={post.keywords} /&gt;
    &lt;/article&gt;
  )
}</code></pre>

<p class="basic-text-box">
  메타데이터 태그는 자동으로 head 색션으로 이동하며 SSR,CSR 모두에서 작동합니다.
</p>

<h3>8-7. 스타일시트 지원</h3>

<p class="basic-text-box">
  스타일시트의 우선순위와 삽입 순서를 관리할 수 있는 기능을 제공합니다.
</p>

<pre class="legacy-code"><code class="language-javascript">function ComponentOne() {
  return (
    &lt;Suspense fallback=&quot;loading...&quot;&gt;
      &lt;link rel=&quot;stylesheet&quot; href=&quot;foo&quot; precedence=&quot;default&quot; /&gt;
      &lt;link rel=&quot;stylesheet&quot; href=&quot;bar&quot; precedence=&quot;high&quot; /&gt;
    &lt;/Suspense&gt;
  )
}</code></pre>

<p class="basic-text-box">
  SSR에서는 head에 포함되고 CSR에서는 스타일 시트가 로드된 후 콘텐츠가 표시되며 중복되는 스타일은 한 번만 삽입됩니다.
</p>

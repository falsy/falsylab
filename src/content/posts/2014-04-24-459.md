소스보기
<pre class="legacy-code"><code class="language-html">[HTML]

&lt;iframe src=&quot;http://player.vimeo.com/video/12345678&quot; id=&quot;vimeo_iframe&quot;&gt;&lt;/iframe&gt;</code></pre>
<pre class="legacy-code"><code class="language-javascript">[JAVASCRIPT]

jQuery(document).ready(function($){
		
	$(window).load(function() {
		cgs_vimeo_volume_control();
		//window가 로드되면 함수 실행
	});
	
	function cgs_vimeo_volume_control() {
	
		var f = $(&quot;#vimeo_iframe&quot;);//&lt;iframe src=&quot;http://player.vimeo.com/video/12345678&quot; id=&quot;vimeo_iframe&quot;&gt;&lt;/iframe&gt;&quot;;
	
		if (!f || !f.attr(&quot;src&quot;)) return;
	
		var url = f.attr(&quot;src&quot;).split(&quot;?&quot;)[0];
	
		var data = {
			&quot;method&quot; : &quot;setVolume&quot;,
			&quot;value&quot; : 0
		};//볼륨을 0으로 설정
	
		f[0].contentWindow.postMessage(JSON.stringify(data), url);
	
		var data = {
			&quot;method&quot; : &quot;play&quot;,
			&quot;value&quot; : 1
		}//영상 재생
	
		f[0].contentWindow.postMessage(JSON.stringify(data), url);
	}

});</code></pre>
<h3>클릭으로 영상 음소거 후 재생 &amp; 일시정지하기</h3>
소스보기
<pre class="legacy-code"><code class="language-html">[HTML]

&lt;iframe src=&quot;http://player.vimeo.com/video/12345678&quot; id=&quot;vimeo_iframe&quot;&gt;&lt;/iframe&gt;

&lt;!-- button --&gt;
&lt;p id=&quot;play&quot;&gt;재생&lt;/p&gt;
&lt;p id=&quot;stop&quot;&gt;일시정지&lt;/p&gt;</code></pre>
<pre class="legacy-code"><code class="language-default">[JAVASCRIPT]

jQuery(document).ready(function($){

	vimeo_play_and_stop();
	
	function vimeo_play_and_stop(){
	
	    var f = $(&quot;#vimeo_iframe&quot;);
	            
	    if(!f || !f.attr(&quot;src&quot;) ) return;
	            
	    var url = f.attr(&quot;src&quot;).split(&quot;?&quot;)[0];
	                           
	    $(&quot;#play&quot;).click(function(){
	        
            var data = {
	            method : &quot;setVolume&quot;,
	            value : 0
	        }
	                
	        f[0].contentWindow.postMessage(JSON.stringify(data), url); 
	            
	        var data = { 
	            method: &quot;play&quot;,
	            value: 1 
	        };
	                
	        f[0].contentWindow.postMessage(JSON.stringify(data), url);
	                
	    });
	            
	    $(&quot;#stop&quot;).click(function(){
	               
	        var data = { 
	            method: &quot;pause&quot;,
	            value: 1 
	        };
	                
	        f[0].contentWindow.postMessage(JSON.stringify(data), url);
	                
	    });
	}

});</code></pre>
